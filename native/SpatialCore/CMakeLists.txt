cmake_minimum_required(VERSION 3.22)
project(AFSpatialCore VERSION 0.1.0 LANGUAGES C CXX)

include(GNUInstallDirs)
include(CTest)
option(AFSC_BUILD_SHARED "Build the C ABI shared library for desktop/replay" ON)
option(AFSC_BUILD_JNI "Build the Java/Android JNI bridge" OFF)
option(AFSC_BUILD_EXAMPLES "Build dependency-free C++ consumer" ON)

get_filename_component(AFSC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(AFSC_APRILTAG "${AFSC_ROOT}/third_party/apriltag")
file(STRINGS "${AFSC_APRILTAG}/sources.txt" AFSC_APRILTAG_RELATIVE_SOURCES)
list(TRANSFORM AFSC_APRILTAG_RELATIVE_SOURCES PREPEND "${AFSC_APRILTAG}/")
file(GLOB AFSC_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
file(GLOB AFSC_HEADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/afsc*.h")
find_package(Threads REQUIRED)

# Build C and C++ once; a C ABI keeps all platform frameworks outside this target.
add_library(afsc_objects OBJECT ${AFSC_SOURCES}
    "${AFSC_ROOT}/native/AprilTag/AFTagCore.c"
    "${AFSC_ROOT}/native/AprilTag/AFTagPoseConventions.c" ${AFSC_APRILTAG_RELATIVE_SOURCES})
set_target_properties(afsc_objects PROPERTIES POSITION_INDEPENDENT_CODE ON
    C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS YES
    CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_include_directories(afsc_objects PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include"
    "${AFSC_ROOT}/native/AprilTag" PRIVATE "${AFSC_APRILTAG}")
target_link_libraries(afsc_objects PRIVATE Threads::Threads)

function(afsc_configure_library target)
    target_include_directories(${target} PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<BUILD_INTERFACE:${AFSC_ROOT}/native/AprilTag>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/afsc>)
    target_link_libraries(${target} PUBLIC Threads::Threads)
    if(UNIX AND NOT APPLE)
        target_link_libraries(${target} PUBLIC m)
    endif()
    set_target_properties(${target} PROPERTIES OUTPUT_NAME af_spatial_core)
endfunction()

add_library(af_spatial_core STATIC $<TARGET_OBJECTS:afsc_objects>)
afsc_configure_library(af_spatial_core)
add_library(AFSpatialCore::core ALIAS af_spatial_core)
set_target_properties(af_spatial_core PROPERTIES EXPORT_NAME core)
set(AFSC_INSTALL_TARGETS af_spatial_core)
if(AFSC_BUILD_SHARED)
    add_library(af_spatial_core_shared SHARED $<TARGET_OBJECTS:afsc_objects>)
    afsc_configure_library(af_spatial_core_shared)
    set_target_properties(af_spatial_core_shared PROPERTIES EXPORT_NAME shared
        VERSION ${PROJECT_VERSION} SOVERSION 1 WINDOWS_EXPORT_ALL_SYMBOLS YES)
    list(APPEND AFSC_INSTALL_TARGETS af_spatial_core_shared)
endif()

if(BUILD_TESTING)
    file(GLOB AFSC_TEST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tests/*_test.cpp")
    foreach(test_source IN LISTS AFSC_TEST_SOURCES)
        get_filename_component(test_name "${test_source}" NAME_WE)
        add_executable(${test_name} "${test_source}")
        target_compile_features(${test_name} PRIVATE cxx_std_17)
        target_link_libraries(${test_name} PRIVATE af_spatial_core)
        target_compile_options(${test_name} PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/UNDEBUG> $<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-UNDEBUG>)
        add_test(NAME ${test_name} COMMAND ${test_name})
    endforeach()
    foreach(smoke IN ITEMS AFTagCoreSmoke AFTagPoseSmoke AFTagPoseConventionsTests)
        add_executable(${smoke} "${AFSC_ROOT}/native/AprilTag/tests/${smoke}.c")
        target_link_libraries(${smoke} PRIVATE af_spatial_core)
        target_include_directories(${smoke} PRIVATE "${AFSC_APRILTAG}")
        target_compile_options(${smoke} PRIVATE $<$<C_COMPILER_ID:MSVC>:/UNDEBUG> $<$<NOT:$<C_COMPILER_ID:MSVC>>:-UNDEBUG>)
        add_test(NAME ${smoke} COMMAND ${smoke})
    endforeach()
endif()

if(AFSC_BUILD_EXAMPLES)
    add_executable(afsc_cpp_example "${AFSC_ROOT}/robot/cpp/example.cpp")
    target_compile_features(afsc_cpp_example PRIVATE cxx_std_17)
    target_link_libraries(afsc_cpp_example PRIVATE af_spatial_core)
    if(BUILD_TESTING)
        add_test(NAME cpp_consumer COMMAND afsc_cpp_example)
    endif()
endif()

if(AFSC_BUILD_JNI)
    set(AFSC_JAVA_SOURCE "${AFSC_ROOT}/robot/java/src/main/java/com/aprilfools/spatial/SpatialCore.java")
    if(ANDROID)
        # Android's toolchain supplies jni.h. This header is generated by javac -h
        # and checked against the Java source by scripts/test-spatial-core.sh.
        set(AFSC_JNI_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/jni/generated")
    else()
        find_package(Java 11 REQUIRED COMPONENTS Development)
        # Cross builds only need the portable JNI declarations, never a host JVM
        # library. Supply the JDK's include + platform include directories explicitly.
        set(AFSC_JNI_INCLUDE_DIRS "" CACHE STRING "JNI header directories for a cross build")
        if(AFSC_JNI_INCLUDE_DIRS)
            set(JNI_INCLUDE_DIRS ${AFSC_JNI_INCLUDE_DIRS})
        else()
            find_package(JNI REQUIRED)
        endif()
        set(AFSC_JNI_INCLUDE "${CMAKE_CURRENT_BINARY_DIR}/jni-generated")
        set(AFSC_JAVA_CLASSES "${CMAKE_CURRENT_BINARY_DIR}/java-classes")
        add_custom_command(OUTPUT "${AFSC_JNI_INCLUDE}/com_aprilfools_spatial_SpatialCore.h"
            COMMAND ${CMAKE_COMMAND} -E make_directory "${AFSC_JNI_INCLUDE}" "${AFSC_JAVA_CLASSES}"
            COMMAND "${Java_JAVAC_EXECUTABLE}" -h "${AFSC_JNI_INCLUDE}" -d "${AFSC_JAVA_CLASSES}" "${AFSC_JAVA_SOURCE}"
            DEPENDS "${AFSC_JAVA_SOURCE}" VERBATIM)
        add_custom_target(afsc_jni_headers DEPENDS "${AFSC_JNI_INCLUDE}/com_aprilfools_spatial_SpatialCore.h")
    endif()
    add_library(af_spatial_core_jni SHARED jni/SpatialCoreJNI.cpp)
    set_target_properties(af_spatial_core_jni PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
    target_include_directories(af_spatial_core_jni PRIVATE "${AFSC_JNI_INCLUDE}" ${JNI_INCLUDE_DIRS})
    target_link_libraries(af_spatial_core_jni PRIVATE af_spatial_core)
    if(NOT ANDROID)
        add_dependencies(af_spatial_core_jni afsc_jni_headers)
        add_custom_command(TARGET af_spatial_core_jni POST_BUILD
            COMMAND "${Java_JAVAC_EXECUTABLE}" -cp "${AFSC_JAVA_CLASSES}" -d "${AFSC_JAVA_CLASSES}"
                "${AFSC_ROOT}/robot/java/src/test/java/com/aprilfools/spatial/SpatialCoreSmoke.java"
            COMMAND "${Java_JAR_EXECUTABLE}" cf "${CMAKE_CURRENT_BINARY_DIR}/af-spatial-core.jar" -C "${AFSC_JAVA_CLASSES}" com/aprilfools/spatial/SpatialCore.class
            VERBATIM)
        if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
            add_test(NAME jni_consumer COMMAND "${Java_JAVA_EXECUTABLE}"
                "-Djava.library.path=$<TARGET_FILE_DIR:af_spatial_core_jni>"
                -cp "${AFSC_JAVA_CLASSES}" com.aprilfools.spatial.SpatialCoreSmoke)
        endif()
        install(FILES "${CMAKE_CURRENT_BINARY_DIR}/af-spatial-core.jar" DESTINATION ${CMAKE_INSTALL_DATADIR}/afsc/java)
    endif()
    list(APPEND AFSC_INSTALL_TARGETS af_spatial_core_jni)
endif()

install(TARGETS ${AFSC_INSTALL_TARGETS} EXPORT AFSpatialCoreTargets
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES ${AFSC_HEADERS} "${AFSC_ROOT}/native/AprilTag/AFTagCore.h"
    "${AFSC_ROOT}/native/AprilTag/AFTagPoseConventions.h"
    DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/afsc)
install(FILES "${AFSC_APRILTAG}/LICENSE.md" "${AFSC_APRILTAG}/NOTICE.md"
    "${AFSC_APRILTAG}/REVISION" DESTINATION ${CMAKE_INSTALL_DATADIR}/afsc/apriltag)
install(EXPORT AFSpatialCoreTargets NAMESPACE AFSpatialCore:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/AFSpatialCore)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/AFSpatialCoreConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/AFSpatialCoreConfig.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/AFSpatialCore)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/AFSpatialCoreConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/AFSpatialCoreConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/AFSpatialCoreConfigVersion.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/AFSpatialCore)
set(CPACK_PACKAGE_NAME AFSpatialCore)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_GENERATOR TGZ)
include(CPack)
